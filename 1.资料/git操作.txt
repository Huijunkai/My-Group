# 切换到你的分支（比如 dev）
git checkout dev

# 语法：git checkout 分支名 -- 文件路径
# 示例1：只拉取单个文件
git checkout main -- src/utils.js

# 示例2：拉取多个文件
git checkout main -- src/utils.js config.json

# 示例3：拉取整个文件夹
git checkout main -- public/

# 示例4：拉取嵌套目录下的文件
git checkout main -- src/components/Header.js

git add .
git commit -m "chore: 从main分支同步指定文件"
